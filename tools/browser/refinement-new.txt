async (page) => {
  page.setDefaultTimeout(7000);
  const checks=[], errors=[];
  const assert=(ok,name)=>{if(!ok)throw Error(name);checks.push(name)};
  const capture=async name=>page.screenshot({path:`docs/visual/captures/refinement-2026-09-27/${name}.png`,scale:'css'});
  page.on('pageerror',e=>errors.push(e.message));
  await page.emulateMedia({reducedMotion:'no-preference'});
  await page.setViewportSize({width:1440,height:1000});
  await page.goto('http://127.0.0.1:4625/'); await page.waitForTimeout(2700);
  await capture('hero-refined');
  await page.locator('#materials').scrollIntoViewIfNeeded(); await page.waitForTimeout(1300);
  await capture('materials-texture');
  assert((await page.locator('.site-header').boundingBox()).y===0,'Header stays available through the long page');
  assert(await page.locator('.reading-progress').evaluate(el=>getComputedStyle(el).transform!=='matrix(0, 0, 0, 1, 0, 0)'),'Reading rule reflects scroll progress');
  await page.getByRole('group',{name:'Explore material notes'}).getByRole('button',{name:/Light/}).click(); await page.waitForTimeout(1200);
  assert((await page.locator('.material-note').innerText()).includes('Let the light do the work'),'Light selector updates the material narrative');
  assert((await page.locator('.material-photo-layer img').getAttribute('src')).endsWith('gallery.webp'),'Light selector shows the matching local reference');
  await capture('materials-light');
  await page.getByRole('button',{name:'Explore this study',exact:true}).click(); await page.waitForTimeout(1100);
  assert((await page.locator('#study-title').innerText())==='Room for a pause','Material note opens its matching study');
  await page.getByRole('button',{name:'Open study 02: The quiet turn',exact:true}).click(); await page.waitForTimeout(1100);
  assert((await page.locator('#study-title').innerText())==='The quiet turn','Gallery thumbnails jump directly to a study');
  assert(await page.getByRole('button',{name:'Open study 02: The quiet turn',exact:true}).getAttribute('aria-pressed')==='true','Gallery selected thumbnail is announced');
  await capture('gallery-thumbnails'); await page.keyboard.press('Escape');
  assert(await page.getByRole('button',{name:'Explore this study',exact:true}).evaluate(el=>el===document.activeElement),'Material study restores its opener focus');
  await page.getByRole('group',{name:'Explore material notes'}).getByRole('button',{name:/Form/}).click(); await page.waitForTimeout(1200); await capture('materials-form');
  await page.locator('.drawing-board').scrollIntoViewIfNeeded(); await page.waitForTimeout(2800); await capture('drawing-plan');
  assert(await page.locator('.plan-line').first().evaluate(el=>el.style.strokeDasharray!==''),'Architectural strokes are controlled by DrawSVG');
  await page.locator('.process-step').nth(2).scrollIntoViewIfNeeded(); await page.waitForTimeout(1000);
  assert(await page.locator('.drawing-board').evaluate(el=>el.classList.contains('drawing-stage-2')),'Plan emphasis follows the final design chapter');
  await capture('drawing-final');
  await page.locator('.brief-progress').scrollIntoViewIfNeeded();
  assert((await page.locator('.brief-progress [role=status]').innerText())==='0 of 4 details','New brief starts with accurate completion feedback');
  await page.getByRole('radio',{name:'A home',exact:true}).check();
  await page.getByRole('radio',{name:'A complete space',exact:true}).check();
  await page.getByRole('radio',{name:'Within a year',exact:true}).check();
  await page.getByRole('button',{name:/Natural light/}).click();
  assert((await page.locator('.brief-progress [role=status]').innerText())==='4 of 4 details','All four choices complete the brief progress');
  await page.locator('.brief-progress').scrollIntoViewIfNeeded(); await page.waitForTimeout(1400); await capture('brief-progress');
  await page.setViewportSize({width:390,height:844});
  await page.locator('.material-choices').scrollIntoViewIfNeeded(); await page.waitForTimeout(800); await capture('material-controls-mobile');
  await page.getByRole('group',{name:'Explore material notes'}).getByRole('button',{name:/Texture/}).click(); await page.waitForTimeout(1200);
  assert((await page.locator('.material-photo-layer img').getAttribute('src')).endsWith('timber.webp'),'Material selection works at phone width');
  await page.locator('.materials-image').scrollIntoViewIfNeeded(); await capture('material-photo-mobile');
  await page.locator('.drawing-board').scrollIntoViewIfNeeded(); await page.waitForTimeout(900); await capture('drawing-mobile');
  assert(await page.evaluate(()=>document.documentElement.scrollWidth<=innerWidth),'New material and drawing layouts fit phone width');
  await page.emulateMedia({reducedMotion:'reduce'}); await page.waitForTimeout(200);
  assert(await page.locator('.plan-line').first().evaluate(el=>el.style.strokeDasharray===''),'Reduced motion restores complete unanimated drawing');
  assert(errors.length===0,'No browser errors in new interactions');
  return {checks,errors,scope:'Actual interaction and sampled states; not uninterrupted viewing.'};
}
